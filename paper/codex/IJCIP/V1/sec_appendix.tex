\section{Further Properties of Budgeted Marginal Risk}\label{app:proofs}
%===============================================================================

\begin{proposition}[Ladder and candidate-set monotonicity]\label{prop:ladder}
\emph{(a)} $M^{(0)}_{i\to y}\le M^{(1)}_{i\to y}\le\cdots\le M^{(|H|-1)}_{i\to y}$;
\emph{(b)} enlarging the candidate set from $H$ to $H'\supseteq H$ cannot decrease $\Mik$;
\emph{(c)} if $f'$ duplicates $f$, i.e.\ $\Vy(B\cup T\cup\{f'\})=\Vy(B\cup T\cup\{f\})=\Vy(B\cup T\cup\{f,f'\})$
for all $T$, deleting $f'$ leaves $\Mik$ unchanged for every $i\ne f,f'$.
\end{proposition}
\begin{proof}
(a) and (b) are maxima over nested feasible sets. (c) If a witness $T$ of $i$ contains $f'$,
replace $f'$ by $f$ (if $f\notin T$) or drop it (if $f\in T$); the value is unchanged and the budget
is respected, so the maximum does not decrease after deleting $f'$, and by (b) it does not increase.
\end{proof}

\begin{theorem}[Truncated super-efficiency]\label{thm:super}
For $A\subseteq H$ with $|A|\le K+1$, $\Vy(B\cup A)-\Vy(B)\le\sum_{i\in A}\Mik$. More tightly, with
$U^{(K)}(\emptyset)=0$ and $U^{(K)}(A)=\min_{j\in A}[U^{(K)}(A\setminus\{j\})+M^{(\min(|A|-1,K))}_{j\to y}]$,
$\Vy(B\cup A)-\Vy(B)\le U^{(K)}(A)\le\sum_{i\in A}\Mik$.
\end{theorem}
\begin{proof}
Order $A=\{i_1,\dots,i_m\}$ and telescope $v_B(A)=\sum_t[v_B(A_t)-v_B(A_{t-1})]$; the $t$-th term is a
marginal of $i_t$ over a background of size $t-1\le K$, hence at most $M^{(K)}_{i_t\to y}$. The staged
bound follows by induction on $|A|$ and Proposition~\ref{prop:ladder}(a).
\end{proof}
For $K\ge|H|-1$ this is the super-efficiency of MCI~\cite{catav2021mci}; with a finite budget it is
stronger because $\Mik\le\mathrm{MCI}$. It yields a release rule ``accept $A$ if $\Vy(B)+U^{(K)}(A)\le\tau$''
that never accepts an unsafe set of size $\le K+1$. We do not use this rule in the main text because
it is conservative for larger release sets; withholding a minimum hitting set of the minimal unsafe sets
(Section~\ref{sec:application}) addresses the same need directly.

\begin{proposition}[Irreducible synergy is bounded by field risk]\label{prop:syn}
With $\mathrm{syn}^B(S)=\Vy(B\cup S)-\max_{T\subsetneq S}\Vy(B\cup T)$ and $|S|\ge 2$,
$\mathrm{syn}^B(S)\le\min_{j\in S}M^{(|S|-1)}_{j\to y}$.
\end{proposition}
\begin{proof}
$\max_{T\subsetneq S}\Vy(B\cup T)\ge\Vy(B\cup S\setminus\{j\})$ for every $j$, hence
$\mathrm{syn}^B(S)\le\Delta_j(B\cup S\setminus\{j\})\le M^{(|S|-1)}_{j\to y}$.
\end{proof}

\begin{remark}[Amplification is weaker than synergy]
$\Mik=M^{(0)}_{i\to y}$ iff $\Delta_i(B\cup T)\le\Delta_i(B)$ for all $|T|\le K$. Submodularity implies no
amplification but not conversely: with $\Vy(\{1\})=0.8$, $\Vy(\{2\})=0.6$, $\Vy(\{3\})=0.8$, all pairs $0.8$
and $\Vy(\{1,2,3\})=1$, every field has $M^{(2)}=M^{(0)}$, yet $\Delta_1(\{3\})=0<\Delta_1(\{2,3\})=0.2$.
\end{remark}

\section{Implementation Details}\label{app:impl}
%===============================================================================

\paragraph{Attackers} Single- and multi-target perceptrons: two hidden layers of 128 ReLU units,
dropout 0.15, Adam (learning rate $10^{-3}$, weight decay $5\times10^{-4}$), batch 128, at most 300
epochs, patience 60, epoch chosen per (set, target) on validation. Many attackers are trained in
parallel on one GPU as independent networks whose inputs are masked to their own field set; masked
columns receive no gradient, so each network is functionally a network trained on its set alone.
Gradient boosting: histogram gradient boosting with (learning rate, leaves) $\in\{(0.05,15),(0.1,31)\}$,
300 iterations with internal early stopping; the configuration is chosen on validation. The family
member is chosen per (set, target) on validation. Across data sets gradient boosting is selected for
35--76\% of (set, target) pairs (RTS-GMLC 76\%, NEM 35\%, PJM 67--69\%, CAISO 52\%), so the family maximum matters.

\paragraph{Estimator} Backbone: three hidden layers of 256 ReLU units with dropout 0.1, input
$[x\odot m, m]$, Adam ($10^{-3}$, weight decay $5\times10^{-4}$), batch 256, at most 400 epochs,
patience 60 on the target loss under eight fixed validation masks. The output layer predicts the targets and all
candidate columns; the reconstruction loss is averaged over hidden columns only. The untrained networks share the
architecture and are initialized with the default PyTorch initialization under seeds 0, 1, 2; seed $r$ of the estimator pairs
the reconstruction backbone and the untrained network of seed $r$. Ridge penalties $\lambda\in\{10^{-3},\dots,10^{2}\}$.
The readout computes $Z^{\top}Z$, $Z^{\top}y$ and the column sums once on all training rows and obtains those of each
cross-validation fold by subtracting the held-out rows; the six penalties are solved as one batched linear system. This
changes only the order of computation (estimates are identical to a direct implementation) and makes a readout about
2.8 times faster; all times reported use it. Queries are processed in batches of eight field sets.

\paragraph{Estimators compared} All use the same standardized data, training rows and random split as the ground truth,
and default settings. \emph{Full model:} a multi-output perceptron with two hidden layers of 128 ReLU units (dropout 0.15),
trained on all candidate fields (Adam, $10^{-3}$, weight decay $5\times10^{-4}$, batch 128, at most 300 epochs, early stopping
with patience 60 on validation rows), three seeds. \emph{Mean substitution:} the removed fields are set to their training
mean (zero after standardization) and the three full models' predictions are averaged. \emph{LazyVI:} the full model of seed 0
is linearized in its parameters at the inputs with removed fields set to zero; the residual $y-f(\tilde x;\theta_0)$ is fitted
by kernel ridge regression with the neural tangent kernel $JJ^{\top}$, one output at a time, with the penalty chosen by exact
leave-one-out cross-validation from $\{10^{-3},\dots,10^{3}\}$ times the mean eigenvalue; the kernel is formed and decomposed in
double precision (single precision was ten times faster but visibly less accurate). \emph{Warm start:} from the full model of
seed 0, at most 100 Adam steps (batch 256) on the inputs with removed fields set to zero, validation loss checked every ten
steps and the best state kept. \emph{Masked surrogate:} the target-only masked backbones' own outputs, averaged over three
seeds. \emph{TabPFN v2:} package \texttt{tabpfn} 2.2.1 with its default ensemble of eight members, one regressor per target,
fitted in context on the training rows of the columns in $S$. Times are measured on one otherwise idle GPU, for all targets of
a data set; the full models take 11--18\,s each to train.

\paragraph{Inference-graph baseline} Following the structure of~\cite{hu2026ignn} without its
expert-assigned intrinsic scores: nodes are the candidate fields and the target; the weight of edge
$i\to j$ is the validation $\Rtwo$ of predicting field $j$ from field $i$ alone by gradient boosting; the
score of $i$ is $\sum_{h=1}^{3}0.5^{\,h-1}(W^{h})_{i,y}$. The first term equals single-field
predictability; longer paths add inference chains through other fields.

\paragraph{SAGE} Marginal imputation with 32 background rows, 512 evaluation rows from the
validation split and 24 random permutations, on a gradient-boosting model fitted to all candidates.

\section{One Backbone for All Target Definitions}\label{app:scope}
In practice the auditor's target list and candidate pool change---a field is reclassified, a new
target is added. With masked reconstruction over all 53 columns of a market, one backbone serves
every such definition through the readout alone. Table~\ref{tab:backbone} compares it with
backbones supervised on each group's own targets. The all-column backbone is more accurate for
$V$ (0.040 versus 0.049) and slightly more accurate for $M^{(2)}$ (0.026 versus 0.029), and its
certified top-1 ratio rises from 0.906 to 0.944.

The mechanism is visible in the representation. A backbone that serves one or two targets leaves a
quarter of its hidden units inactive (24\%) and has an effective dimension of 1.45; one that must
reconstruct every column keeps 92\% of units active and has an effective dimension of 2.44. It has
to encode the latent operating state---load level, renewable output, congestion---that drives all
grid quantities, as argued for multi-task representation
learning~\cite{caruana1997multitask,baxter2000inductive,maurer2016benefit} (Fig.~\ref{fig:scope}). Why $V$ improves more
than $M$ follows from~\eqref{eq:prop}: 89--98\% of the $V$ error is common-mode, and a better
representation mostly shrinks this common part, which cancels in the marginals.

The last row of Table~\ref{tab:backbone} tests a target never seen in pre-training: the target
column is deleted from the data before pre-training. Set values and marginals are estimated as well
as with the all-column backbone, but $\hat M^{(2)}$ is inflated (bias $+0.028$), because the
maximum over many marginals picks up the few that are overestimated; the certified top-1 ratio drops
to 0.75. Retraining three backgrounds per field restores it to 0.966. Re-use for unseen targets
therefore requires certification, which the scan--certify pipeline provides.

\section{A Synthetic Example}\label{app:toy}
%===============================================================================

Fig.~\ref{fig:mechanism} uses synthetic data with a known structure,
$Y=a\,AB+bP+\sigma\varepsilon$, where $A$ and $B$ act only jointly, $C_1$ and $C_2$ are two
noisy copies of $P$, $D$ is a weak copy of $P$ and $N$ is noise; all 63 sets are retrained.
Correlation, single-field retraining and pairwise inference-graph propagation give $A$ and $B$
a score of zero although $\{A,B\}$ reaches $0.54$. LOCO gives $C_1$ and $C_2$ almost zero because
each substitutes for the other, and LOCO, permutation importance and SAGE all give $D$ zero
because the full model relies on $C_1,C_2$ instead---yet $\{A,B,D\}$ crosses $\tau=0.7$ while
$\{A,B\}$ does not, so $D$ is critical. SAGE splits credit within both pairs. $M^{(2)}$ assigns
$A$, $B$, $C_1$, $C_2$ their full worst-case contribution and gives $D$ a non-zero risk with the
witness $\{A,B\}$. This is the behaviour predicted by Table~\ref{tab:req}: averages dilute,
single contexts miss substitutes, and pairwise edges miss joint inference.

\section{Estimator Design Diagnostics}\label{app:diagnostics}
The comparisons below examine representation, seed aggregation and numerical stability under the existing experimental protocol. They supplement the principal configurations in Table~\ref{tab:design-main}.

\paragraph{Representation}
Target-only masked pre-training yields an effective feature dimension of 1.60, compared with 3.30 for an untrained network. Restricting masks to one to four visible fields gives 1.65. Reconstruction raises the dimension to 2.75. At two retained principal components, trained features have about half the set-value error of random features, but the remaining dimensions supply much of the small-set readout's accuracy. For 16-field queries, random features have error 0.074, compared with 0.027 for target-only and 0.023 for reconstruction features; this larger-set diagnostic uses a gradient-boosting reference and is separate from the exhaustive main audit.

\paragraph{Seeds and numerical stability}
Averaging three predictions reduces the full estimator's set-value error from 0.020 to 0.018. Concatenating the features of three target-only seeds into one readout costs 41\,ms, compared with 21\,ms for averaging their separate predictions, and yields a lower top-1 certified ratio on the six targets used in that comparison (0.916 versus 0.953). Rare unstable predictions disproportionately affect the maximum marginal gain: without prediction clipping, the reconstruction configuration has risk error 0.027 and certified ratio 0.881, despite its 0.020 set-value error. The safeguards in Section~\ref{sec:model} address this failure. Uniform visible-count and Bernoulli masks give similar risk errors (0.023 and 0.024); eliminating masks raises the error to 0.036.

\section{Population Reference and Estimator Analysis}\label{app:estimator-analysis}
\begin{definition}[Population inferability]
\[
\Vy^{*}(S) = 1 - \inf_{f}\,\mathbb{E}\big[(Y_y - f(X_S))^{2}\big] \big/ \mathrm{Var}(Y_y),
\]
the infimum taken over measurable $f$. It satisfies $\Vy^{*}(\emptyset)=0$, $0 \le \Vy^{*} \le 1$
and is monotone in $S$.
\end{definition}

\subsection{Why the readout is solved per subset}
A cheaper alternative lets the network output the prediction directly, $\hat y=f_\theta(x\odot
m_S, m_S)$, and reads $\hat V$ off it. Then one set of output weights serves every
conditioning pattern, i.e.\ the attacker is forced to combine every subset of fields in the
same way---the very effect that combination risk is about. Solving the readout per subset
keeps the expensive representation shared and the combination rule specific.

\subsection{Error propagation}
Write the estimate as $\hat V(S)=V(S)+b+\varepsilon(S)$, where $b$ is a bias common to the sets
of interest and $\varepsilon(S)$ the set-specific part. Since a marginal is a difference of two
nested sets, $\hat\Delta_i(T)-\Delta_i(T)=\varepsilon(T\cup\{i\})-\varepsilon(T)$ and
\begin{equation}\label{eq:prop}
\big|\hat M^{(K)}_{i\to y} - \Mik\big| \;\le\; \max_{|T|\le K}\big|\varepsilon(T\cup\{i\})-\varepsilon(T)\big|,
\end{equation}
independently of $b$. A readout on a frozen representation typically underestimates the fully
trained attacker by a similar amount on nested sets, which is exactly a common-mode $b$; this is
why an estimator can be biased in $V$ yet accurate in $M$. The quantity to control is the
\emph{marginal} error, which we report throughout.
